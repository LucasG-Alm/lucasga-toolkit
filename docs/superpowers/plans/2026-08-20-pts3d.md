# PTS3D Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Comando de AutoCAD que converte pontos cotados selecionados em uma 3D Polyline ordenada e/ou um CSV para o Revit.

**Architecture:** Uma DLL única (`LucasGA.Toolkit`). O núcleo (`Nucleo/`) opera apenas sobre listas de `PontoCotado` e não referencia a API do AutoCAD, o que permite testá-lo com o SDK 8 já instalado. A camada de fronteira (`LeitorDePontos`, `DesenhistaPolyline`) e a UI concentram a dependência do AutoCAD e exigem o SDK 10.

**Tech Stack:** VB.NET · .NET 10 (`net10.0-windows`) para a DLL · .NET 8 para o projeto de testes · xUnit · WinForms (`DataGridView` em `PaletteSet`) · AutoCAD 2027 API

## Global Constraints

- Linguagem: **VB.NET**. Não usar C# em nenhum arquivo.
- `Option Strict On` e `Option Explicit On` em todos os arquivos.
- Idioma de identificadores, comentários e mensagens ao usuário: **português**.
- A DLL final é **única** — todos os comandos do toolkit compilam em `LucasGA.Toolkit.dll`.
- Arquivos em `Nucleo/` **não podem** ter `Imports Autodesk.*`. Essa é a regra que mantém o núcleo testável.
- CSV sempre com **ponto** como separador decimal e 3 casas (`CultureInfo.InvariantCulture`).
- O projeto de testes usa `<RootNamespace></RootNamespace>` vazio — sem isso o VB prefixa os namespaces e os `Imports` quebram.
- Nenhum DWG, DXF ou CSV pode ser commitado. `.gitignore` já bloqueia; arquivos de teste vão em `local/`.
- Faixa de tolerância para comparação de cota: **0,001 m**.

```vb
Imports System.Globalization

Namespace Nucleo

    ''' <summary>Sinalizações de problema detectadas num ponto. Combináveis.</summary>
    <Flags>
    Public Enum Alerta
        Nenhum = 0
        DivergenciaCota = 1
        ZSuspeito = 2
        DuplicadoXY = 4
        SaltoNaCadeia = 8
    End Enum

    ''' <summary>
    ''' Um ponto cotado lido do desenho. O Z da geometria é a fonte da cota;
    ''' CotaAtributo é apenas o rótulo gravado no bloco.
    ''' </summary>
    Public Class PontoCotado

        Public Property Handle As String = String.Empty
        Public Property X As Double
        Public Property Y As Double
        Public Property Z As Double
        Public Property Layer As String = String.Empty
        Public Property Numero As String = String.Empty
        Public Property Descricao As String = String.Empty
        Public Property CotaAtributo As Double?
        Public Property OrdemSelecao As Integer
        Public Property Alertas As Alerta = Alerta.Nenhum
        Public Property Marcado As Boolean = True

        ''' <summary>
        ''' Converte o texto de um atributo em número. Os blocos do levantamento
        ''' gravam vírgula decimal ("721,625"); alguns gravam ponto.
        ''' Devolve Nothing quando o texto não é numérico.
        ''' </summary>
        Public Shared Function LerCota(texto As String) As Double?
            If String.IsNullOrWhiteSpace(texto) Then Return Nothing

            Dim limpo As String = texto.Trim().Replace(" ", "")
            Dim valor As Double

            If Double.TryParse(limpo, NumberStyles.Float, CultureInfo.InvariantCulture, valor) Then
                Return valor
            End If

            limpo = limpo.Replace(","c, "."c)
            If Double.TryParse(limpo, NumberStyles.Float, CultureInfo.InvariantCulture, valor) Then
                Return valor
            End If

            Return Nothing
        End Function

    End Class

End Namespace
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports System.Linq

Namespace Nucleo

    Public Enum ModoOrdenacao
        OrdemSelecao = 0
        VizinhoMaisProximo = 1
        Manual = 2
    End Enum

    ''' <summary>
    ''' Define a sequência dos vértices da polilinha. Não conhece o AutoCAD.
    ''' </summary>
    Public Module Ordenador

        ''' <summary>Devolve uma nova lista na ordem determinada pelo modo.</summary>
        Public Function Ordenar(pontos As List(Of PontoCotado), modo As ModoOrdenacao) As List(Of PontoCotado)
            If pontos Is Nothing OrElse pontos.Count <= 1 Then
                Return If(pontos, New List(Of PontoCotado)()).ToList()
            End If

            Select Case modo
                Case ModoOrdenacao.OrdemSelecao
                    Return pontos.OrderBy(Function(p) p.OrdemSelecao).ToList()
                Case ModoOrdenacao.VizinhoMaisProximo
                    Return EncadearPorProximidade(pontos)
                Case Else
                    Return pontos.ToList()
            End Select
        End Function

        ''' <summary>
        ''' Encadeamento guloso: começa no primeiro ponto da seleção e, a cada
        ''' passo, salta para o mais próximo ainda não usado (distância em XY).
        ''' </summary>
        Private Function EncadearPorProximidade(pontos As List(Of PontoCotado)) As List(Of PontoCotado)
            Dim restantes As New List(Of PontoCotado)(pontos)
            Dim atual As PontoCotado = restantes.OrderBy(Function(p) p.OrdemSelecao).First()
            restantes.Remove(atual)

            Dim cadeia As New List(Of PontoCotado) From {atual}

            While restantes.Count > 0
                Dim proximo As PontoCotado = restantes(0)
                Dim menor As Double = DistanciaXY(atual, proximo)

                For i As Integer = 1 To restantes.Count - 1
                    Dim d As Double = DistanciaXY(atual, restantes(i))
                    If d < menor Then
                        menor = d
                        proximo = restantes(i)
                    End If
                Next

                cadeia.Add(proximo)
                restantes.Remove(proximo)
                atual = proximo
            End While

            Return cadeia
        End Function

        ''' <summary>
        ''' Marca com SaltoNaCadeia todo ponto cuja distância ao anterior seja
        ''' maior que fator vezes a distância mediana da cadeia. Sintoma de
        ''' encadeamento que pulou para outra feição.
        ''' Devolve quantos pontos foram marcados.
        ''' </summary>
        Public Function MarcarSaltos(pontos As List(Of PontoCotado), fator As Double) As Integer
            If pontos Is Nothing OrElse pontos.Count < 3 Then Return 0

            Dim distancias As New List(Of Double)()
            For i As Integer = 1 To pontos.Count - 1
                distancias.Add(DistanciaXY(pontos(i - 1), pontos(i)))
            Next

            Dim mediana As Double = Medianar(distancias)
            If mediana <= 0.0 Then Return 0

            Dim limite As Double = mediana * fator
            Dim marcados As Integer = 0

            For i As Integer = 1 To pontos.Count - 1
                If distancias(i - 1) > limite Then
                    pontos(i).Alertas = pontos(i).Alertas Or Alerta.SaltoNaCadeia
                    marcados += 1
                End If
            Next

            Return marcados
        End Function

        Public Function DistanciaXY(a As PontoCotado, b As PontoCotado) As Double
            Dim dx As Double = a.X - b.X
            Dim dy As Double = a.Y - b.Y
            Return Math.Sqrt(dx * dx + dy * dy)
        End Function

        ''' <summary>Mediana de uma lista de valores. Lista vazia devolve 0.</summary>
        Public Function Medianar(valores As List(Of Double)) As Double
            If valores Is Nothing OrElse valores.Count = 0 Then Return 0.0
            Dim v As List(Of Double) = valores.OrderBy(Function(d) d).ToList()
            Dim meio As Integer = v.Count \ 2
            If v.Count Mod 2 = 1 Then Return v(meio)
            Return (v(meio - 1) + v(meio)) / 2.0
        End Function

    End Module

End Namespace
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports System.Linq

Namespace Nucleo

    ''' <summary>
    ''' Insere vértices interpolados entre pontos consecutivos. Usado apenas na
    ''' exportação CSV — a polilinha desenhada preserva os pontos medidos.
    ''' </summary>
    Public Module Densificador

        Public Const DescricaoInterpolado As String = "interpolado"

        ''' <summary>
        ''' Interpola linearmente X, Y e Z ao longo de cada segmento.
        ''' passoMetros menor ou igual a zero devolve a lista original.
        ''' incluirOriginais=False descarta os vértices medidos, mantendo apenas
        ''' os interpolados e as extremidades de cada segmento.
        ''' </summary>
        Public Function Densificar(pontos As List(Of PontoCotado),
                                   passoMetros As Double,
                                   incluirOriginais As Boolean) As List(Of PontoCotado)

            If pontos Is Nothing OrElse pontos.Count < 2 Then
                Return If(pontos, New List(Of PontoCotado)()).ToList()
            End If

            If passoMetros <= 0.0 Then Return pontos.ToList()

            Dim saida As New List(Of PontoCotado)()

            For i As Integer = 0 To pontos.Count - 2
                Dim a As PontoCotado = pontos(i)
                Dim b As PontoCotado = pontos(i + 1)

                    Dim t As Double = (k * passoMetros) / d
                    If t >= 1.0 Then Exit For
                    saida.Add(New PontoCotado With {
                        .X = a.X + (b.X - a.X) * t,
                        .Y = a.Y + (b.Y - a.Y) * t,
                        .Z = a.Z + (b.Z - a.Z) * t,
                        .Layer = a.Layer,
                        .Descricao = DescricaoInterpolado
                    })
                Next
            Next

            saida.Add(pontos(pontos.Count - 1))
            Return saida
        End Function
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports System.Linq

Namespace Nucleo

    ''' <summary>Intervalo de cota considerado plausível para a seleção.</summary>
    Public Class FaixaDeCota
        Public Property Minimo As Double
        Public Property Maximo As Double
    End Class

    ''' <summary>
    ''' Sinaliza problemas nos pontos. Nunca corrige, nunca descarta —
    ''' o usuário vê e decide.
    ''' </summary>
    Public Module AnalisadorDeAlertas

        ''' <summary>Diferença máxima aceita entre Z e atributo de cota, em metros.</summary>
        Public Const ToleranciaCota As Double = 0.001

        ''' <summary>Distância máxima em XY para considerar dois pontos coincidentes.</summary>
        Public Const ToleranciaXY As Double = 0.001

        ''' <summary>
        ''' Faixa plausível a partir da própria nuvem: mediana ± 3× o intervalo
        ''' interquartil. Resiste a outlier grosseiro porque nem mediana nem IQR
        ''' se deslocam com valores extremos isolados.
        ''' Com menos de 4 pontos, devolve faixa aberta.
        ''' </summary>
        Public Function CalcularFaixaPlausivel(pontos As List(Of PontoCotado)) As FaixaDeCota
            If pontos Is Nothing OrElse pontos.Count < 4 Then
                Return New FaixaDeCota With {.Minimo = Double.MinValue, .Maximo = Double.MaxValue}
            End If

            Dim z As List(Of Double) = pontos.Select(Function(p) p.Z).OrderBy(Function(v) v).ToList()
            Dim q1 As Double = Percentil(z, 0.25)
            Dim q3 As Double = Percentil(z, 0.75)
            Dim mediana As Double = Percentil(z, 0.5)
            Dim iqr As Double = q3 - q1

            If iqr <= 0.0 Then
                Return New FaixaDeCota With {.Minimo = mediana - 1.0, .Maximo = mediana + 1.0}
            End If

            Return New FaixaDeCota With {
                .Minimo = mediana - 3.0 * iqr,
                .Maximo = mediana + 3.0 * iqr
            }
        End Function

        ''' <summary>
        ''' Aplica todos os alertas à lista. Devolve quantos pontos ficaram com
        ''' pelo menos um alerta. Preserva alertas já existentes (SaltoNaCadeia).
        ''' </summary>
        Public Function Analisar(pontos As List(Of PontoCotado), faixa As FaixaDeCota) As Integer
            If pontos Is Nothing OrElse pontos.Count = 0 Then Return 0

            For Each p As PontoCotado In pontos
                p.Alertas = p.Alertas And Not (Alerta.DivergenciaCota Or Alerta.ZSuspeito Or Alerta.DuplicadoXY)

                If p.CotaAtributo.HasValue AndAlso
                   Math.Abs(p.Z - p.CotaAtributo.Value) > ToleranciaCota Then
                    p.Alertas = p.Alertas Or Alerta.DivergenciaCota
                End If

                If p.Z = 0.0 OrElse p.Z < faixa.Minimo OrElse p.Z > faixa.Maximo Then
                    p.Alertas = p.Alertas Or Alerta.ZSuspeito
                End If
            Next

            MarcarDuplicidadeXY(pontos)

            Return pontos.Count(Function(p) p.Alertas <> Alerta.Nenhum)
        End Function

        ''' <summary>
        ''' Marca pares no mesmo XY com Z diferente — assinatura de topo e pé de
        ''' muro. Ligá-los numa polilinha cria uma parede vertical dentro de uma
        ''' linha de terreno.
        ''' </summary>
        Private Sub MarcarDuplicidadeXY(pontos As List(Of PontoCotado))
            For i As Integer = 0 To pontos.Count - 2
                For j As Integer = i + 1 To pontos.Count - 1
                    Dim dx As Double = Math.Abs(pontos(i).X - pontos(j).X)
                    Dim dy As Double = Math.Abs(pontos(i).Y - pontos(j).Y)
                    If dx <= ToleranciaXY AndAlso dy <= ToleranciaXY Then
                        If Math.Abs(pontos(i).Z - pontos(j).Z) > ToleranciaCota Then
                            pontos(i).Alertas = pontos(i).Alertas Or Alerta.DuplicadoXY
                            pontos(j).Alertas = pontos(j).Alertas Or Alerta.DuplicadoXY
                        End If
                    End If
                Next
            Next
        End Sub

        ''' <summary>Percentil por interpolação linear. Lista deve vir ordenada.</summary>
        Private Function Percentil(ordenados As List(Of Double), fracao As Double) As Double
            If ordenados.Count = 1 Then Return ordenados(0)
            Dim pos As Double = fracao * (ordenados.Count - 1)
            Dim baixo As Integer = CInt(Math.Floor(pos))
            Dim alto As Integer = CInt(Math.Ceiling(pos))
            If baixo = alto Then Return ordenados(baixo)
            Return ordenados(baixo) + (ordenados(alto) - ordenados(baixo)) * (pos - baixo)
        End Function

    End Module

End Namespace
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports System.Globalization
Imports System.IO
Imports System.Linq
Imports System.Text

Namespace Nucleo

    ''' <summary>Configuração da exportação para o Revit.</summary>
    Public Class OpcoesCsv
        ''' <summary>Easting subtraído de cada X. O Revit perde precisão longe da origem.</summary>
        Public Property OrigemE As Double = 0.0
        ''' <summary>Northing subtraído de cada Y.</summary>
        Public Property OrigemN As Double = 0.0
        Public Property Densificar As Boolean = False
        Public Property PassoMetros As Double = 0.5
        Public Property IncluirOriginais As Boolean = True
    End Class

    ''' <summary>
    ''' Escreve o arquivo de pontos no formato que o Revit espera em
    ''' Toposolid → Create from Import → Points File: X,Y,Z sem cabeçalho.
    ''' </summary>
    Public Module ExportadorCsv

        ''' <summary>
        ''' Produz as linhas do arquivo. Considera apenas pontos marcados,
        ''' na ordem em que chegam.
        ''' </summary>
        Public Function GerarLinhas(pontos As List(Of PontoCotado), opcoes As OpcoesCsv) As List(Of String)
            Dim linhas As New List(Of String)()
            If pontos Is Nothing OrElse pontos.Count = 0 Then Return linhas

            Dim op As OpcoesCsv = If(opcoes, New OpcoesCsv())
            Dim usados As List(Of PontoCotado) = pontos.Where(Function(p) p.Marcado).ToList()
            If usados.Count = 0 Then Return linhas

            If op.Densificar AndAlso op.PassoMetros > 0.0 Then
                usados = Densificador.Densificar(usados, op.PassoMetros, op.IncluirOriginais)
            End If

            Dim ci As CultureInfo = CultureInfo.InvariantCulture
            For Each p As PontoCotado In usados
                linhas.Add(String.Format(ci, "{0:F3},{1:F3},{2:F3}",
                                         p.X - op.OrigemE,
                                         p.Y - op.OrigemN,
                                         p.Z))
            Next

            Return linhas
        End Function

        ''' <summary>
        ''' Grava o arquivo em UTF-8 sem BOM. Devolve o número de linhas escritas.
        ''' Exceções de E/S sobem para quem chamou tratar.
        ''' </summary>
        Public Function Salvar(pontos As List(Of PontoCotado), opcoes As OpcoesCsv, caminho As String) As Integer
            Dim linhas As List(Of String) = GerarLinhas(pontos, opcoes)
            File.WriteAllLines(caminho, linhas, New UTF8Encoding(False))
            Return linhas.Count
        End Function

    End Module

End Namespace
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports Autodesk.AutoCAD.ApplicationServices
Imports Autodesk.AutoCAD.DatabaseServices
Imports Autodesk.AutoCAD.EditorInput
Imports Nucleo

Namespace Comandos.Pts3d

    Public Class ResultadoLeitura
        Public Property Pontos As New List(Of PontoCotado)()
        Public Property Descartados As Integer
        Public Property Cancelado As Boolean
    End Class

    ''' <summary>
    ''' Fronteira com o AutoCAD: converte a seleção do usuário em dados puros.
    ''' Aceita DBPoint e BlockReference; qualquer outro tipo é contado como
    ''' descartado e reportado.
    ''' </summary>
    Public Module LeitorDePontos

        Public Function Ler(doc As Document) As ResultadoLeitura
            Dim res As New ResultadoLeitura()
            Dim ed As Editor = doc.Editor

            Dim opcoes As New PromptSelectionOptions()
            opcoes.MessageForAdding = vbLf & "Selecione os pontos cotados"
            Dim sel As PromptSelectionResult = ed.GetSelection(opcoes)

            If sel.Status <> PromptStatus.OK Then
                res.Cancelado = True
                Return res
            End If

            Dim ordem As Integer = 0

            Using tr As Transaction = doc.Database.TransactionManager.StartTransaction()
                For Each id As ObjectId In sel.Value.GetObjectIds()
                    Dim ent As Entity = TryCast(tr.GetObject(id, OpenMode.ForRead), Entity)
                    If ent Is Nothing Then Continue For

                    Dim ponto As PontoCotado = Nothing
                    Dim dbp As DBPoint = TryCast(ent, DBPoint)

                    If dbp IsNot Nothing Then
                        ponto = New PontoCotado With {
                            .Handle = dbp.Handle.ToString(),
                            .X = dbp.Position.X, .Y = dbp.Position.Y, .Z = dbp.Position.Z,
                            .Layer = dbp.Layer
                        }
                    Else
                        Dim blk As BlockReference = TryCast(ent, BlockReference)
                        If blk IsNot Nothing Then
                            ponto = New PontoCotado With {
                                .Handle = blk.Handle.ToString(),
                                .X = blk.Position.X, .Y = blk.Position.Y, .Z = blk.Position.Z,
                                .Layer = blk.Layer
                            }
                            LerAtributos(tr, blk, ponto)
                        End If
                    End If

                    If ponto Is Nothing Then
                        res.Descartados += 1
                    Else
                        ponto.OrdemSelecao = ordem
                        ordem += 1
                        res.Pontos.Add(ponto)
                    End If
                Next
                tr.Commit()
            End Using

            Return res
        End Function

        ''' <summary>
        ''' Lê PONTO, DESC e COTA/Z dos atributos. Nomes de tag são comparados
        ''' sem diferenciar maiúsculas.
        ''' </summary>
        Private Sub LerAtributos(tr As Transaction, blk As BlockReference, ponto As PontoCotado)
            For Each attId As ObjectId In blk.AttributeCollection
                Dim att As AttributeReference = TryCast(tr.GetObject(attId, OpenMode.ForRead), AttributeReference)
                If att Is Nothing Then Continue For

                Select Case att.Tag.Trim().ToUpperInvariant()
                    Case "PONTO"
                        ponto.Numero = att.TextString
                    Case "DESC"
                        ponto.Descricao = att.TextString
                    Case "COTA", "Z"
                        Dim v As Double? = PontoCotado.LerCota(att.TextString)
                        If v.HasValue Then ponto.CotaAtributo = v
                End Select
            Next
        End Sub

    End Module

End Namespace
```

```vb
Option Strict On
Option Explicit On

Imports System.Collections.Generic
Imports System.Linq
Imports Autodesk.AutoCAD.ApplicationServices
Imports Autodesk.AutoCAD.DatabaseServices
Imports Autodesk.AutoCAD.Geometry
Imports Nucleo

Namespace Comandos.Pts3d

    ''' <summary>
    ''' Cria a Polyline3d. Tudo numa transação só, então um U desfaz inteiro.
    ''' </summary>
    Public Module DesenhistaPolyline

        ''' <summary>
        ''' Desenha a polilinha com os pontos marcados, na ordem recebida.
        ''' Devolve ObjectId.Null se houver menos de 2 pontos marcados.
        ''' nomeLayer vazio usa a layer corrente; layer inexistente é criada.
        ''' </summary>
        Public Function Desenhar(doc As Document,
                                 pontos As List(Of PontoCotado),
                                 nomeLayer As String) As ObjectId

            Dim usados As List(Of PontoCotado) = pontos.Where(Function(p) p.Marcado).ToList()
            If usados.Count < 2 Then Return ObjectId.Null

            Dim db As Database = doc.Database
            Dim idPolilinha As ObjectId = ObjectId.Null

            Using tr As Transaction = db.TransactionManager.StartTransaction()
                Dim bt As BlockTable = CType(tr.GetObject(db.BlockTableId, OpenMode.ForRead), BlockTable)
                Dim btr As BlockTableRecord = CType(tr.GetObject(bt(BlockTableRecord.ModelSpace), OpenMode.ForWrite), BlockTableRecord)

                Dim pl As New Polyline3d()
                If Not String.IsNullOrWhiteSpace(nomeLayer) Then
                    pl.Layer = GarantirLayer(tr, db, nomeLayer)
                End If

                btr.AppendEntity(pl)
                tr.AddNewlyCreatedDBObject(pl, True)

                For Each p As PontoCotado In usados
                    Dim v As New PolylineVertex3d(New Point3d(p.X, p.Y, p.Z))
                    pl.AppendVertex(v)
                    tr.AddNewlyCreatedDBObject(v, True)
                Next

                idPolilinha = pl.ObjectId
                tr.Commit()
            End Using

            Return idPolilinha
        End Function

        ''' <summary>Cria a layer se não existir. Devolve o nome para atribuição.</summary>
        Private Function GarantirLayer(tr As Transaction, db As Database, nome As String) As String
            Dim lt As LayerTable = CType(tr.GetObject(db.LayerTableId, OpenMode.ForRead), LayerTable)
            If lt.Has(nome) Then Return nome

            lt.UpgradeOpen()
            Dim ltr As New LayerTableRecord()
            ltr.Name = nome
            lt.Add(ltr)
            tr.AddNewlyCreatedDBObject(ltr, True)
            Return nome
        End Function

    End Module

End Namespace
```

```vb
        ''' <summary>Pontos na ordem visível da grade — é o que será desenhado.</summary>
        Public Function PontosNaOrdemExibida() As List(Of PontoCotado)
            Return _pontos.ToList()
        End Function

        Private Sub Redesenhar()
            _grade.Rows.Clear()
            For i As Integer = 0 To _pontos.Count - 1
                Dim p As PontoCotado = _pontos(i)
                Dim delta As String = ""
                If p.CotaAtributo.HasValue Then delta = (p.Z - p.CotaAtributo.Value).ToString("F3")

                Dim idx As Integer = _grade.Rows.Add(
                    p.Marcado, (i + 1).ToString(), p.Numero, p.Descricao, p.Layer,
                    p.X.ToString("F3"), p.Y.ToString("F3"), p.Z.ToString("F3"),
                    If(p.CotaAtributo.HasValue, p.CotaAtributo.Value.ToString("F3"), ""),
                    delta)

                _grade.Rows(idx).DefaultCellStyle.BackColor = CorDoAlerta(p.Alertas)
            Next
        End Sub

        ''' <summary>Vermelho manda sobre laranja, que manda sobre amarelo.</summary>
        Private Shared Function CorDoAlerta(a As Alerta) As Color
            If (a And Alerta.DivergenciaCota) = Alerta.DivergenciaCota Then Return Color.FromArgb(255, 205, 205)
            If (a And Alerta.DuplicadoXY) = Alerta.DuplicadoXY Then Return Color.FromArgb(255, 228, 196)
            If (a And Alerta.ZSuspeito) = Alerta.ZSuspeito Then Return Color.FromArgb(255, 250, 190)
            If (a And Alerta.SaltoNaCadeia) = Alerta.SaltoNaCadeia Then Return Color.FromArgb(220, 235, 255)
            Return Color.White
        End Function

        Private Sub AoMudarCelula(sender As Object, e As DataGridViewCellEventArgs)
            If e.RowIndex < 0 OrElse e.RowIndex >= _pontos.Count Then Return
            If _grade.Columns(e.ColumnIndex).Name <> "Marcado" Then Return
            _pontos(e.RowIndex).Marcado = CBool(_grade.Rows(e.RowIndex).Cells("Marcado").Value)
        End Sub

        Private Sub AoPressionar(sender As Object, e As MouseEventArgs)

            _linhaArrastada = hit.RowIndex
            If _linhaArrastada >= 0 Then
                _grade.DoDragDrop(_grade.Rows(_linhaArrastada), DragDropEffects.Move)
            End If
        End Sub

        Private Sub AoArrastarSobre(sender As Object, e As DragEventArgs)
            e.Effect = DragDropEffects.Move
        End Sub

        Private Sub AoSoltar(sender As Object, e As DragEventArgs)
            Dim ponto As Point = _grade.PointToClient(New Point(e.X, e.Y))
            Dim destino As Integer = _grade.HitTest(ponto.X, ponto.Y).RowIndex
            If _linhaArrastada < 0 OrElse destino < 0 OrElse destino = _linhaArrastada Then Return

            Dim p As PontoCotado = _pontos(_linhaArrastada)
            _pontos.RemoveAt(_linhaArrastada)
            _pontos.Insert(destino, p)
            Redesenhar()
            RaiseEvent OrdemAlterada()
        End Sub

    End Class

End Namespace
```
